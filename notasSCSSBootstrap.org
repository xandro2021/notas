import fs from "fs";
import path from "path";
import { globSync } from "glob";
import sharp from "sharp";

export default function imagenesPlugin() {

  return {

    name: "vite-imagenes",

    async closeBundle() {

      const srcDir = "src/img";
      const buildDir = "public/build/img";

      const images = globSync(`${srcDir}/**/*`);

      const tasks = images.map(async (file) => {

        const relativePath = path.relative(srcDir, path.dirname(file));
        const outputDir = path.join(buildDir, relativePath);

        if (!fs.existsSync(outputDir)) {
          fs.mkdirSync(outputDir, { recursive: true });
        }

        const baseName = path.basename(file, path.extname(file));
        const extName = path.extname(file).toLowerCase();

        if (extName === ".svg") {

          const outputFile = path.join(outputDir, `${baseName}${extName}`);

          if (!fs.existsSync(outputFile)) {
            fs.copyFileSync(file, outputFile);
          }

          return;
        }

        const outputFile = path.join(outputDir, `${baseName}${extName}`);
        const outputWebp = path.join(outputDir, `${baseName}.webp`);
        const outputAvif = path.join(outputDir, `${baseName}.avif`);

        const options = { quality: 80 };

        const promises = [];

        if (!fs.existsSync(outputFile)) {
          promises.push(sharp(file).jpeg(options).toFile(outputFile));
        }

        if (!fs.existsSync(outputWebp)) {
          promises.push(sharp(file).webp(options).toFile(outputWebp));
        }

        if (!fs.existsSync(outputAvif)) {
          promises.push(sharp(file).avif(options).toFile(outputAvif));
        }

        await Promise.all(promises);

        console.log("Imagen procesada:", file);

      });

      await Promise.all(tasks);

      console.log("✓ Todas las imágenes procesadas");

    }

  };

}

import fs from "fs";
import path from "path";
import { globSync } from "glob";
import sharp from "sharp";

export default function imagenesPlugin() {

  return {

    name: "vite-imagenes",

    async closeBundle() {

      const srcDir = "src/img";
      const buildDir = "public/build/img";

      const images = globSync(`${srcDir}/**/*`);

      const sizes = [400, 800, 1200];

      const tasks = images.map(async (file) => {

        const relativePath = path.relative(srcDir, path.dirname(file));
        const outputDir = path.join(buildDir, relativePath);

        if (!fs.existsSync(outputDir)) {
          fs.mkdirSync(outputDir, { recursive: true });
        }

        const baseName = path.basename(file, path.extname(file));
        const extName = path.extname(file).toLowerCase();

        if (extName === ".svg") {

          const outputFile = path.join(outputDir, `${baseName}${extName}`);

          if (!fs.existsSync(outputFile)) {
            fs.copyFileSync(file, outputFile);
          }

          return;
        }

        const options = { quality: 80 };

        const promises = [];

        for (const size of sizes) {

          const webp = path.join(outputDir, `${baseName}-${size}.webp`);
          const avif = path.join(outputDir, `${baseName}-${size}.avif`);

          if (!fs.existsSync(webp)) {
            promises.push(
              sharp(file)
                .resize(size)
                .webp(options)
                .toFile(webp)
            );
          }

          if (!fs.existsSync(avif)) {
            promises.push(
              sharp(file)
                .resize(size)
                .avif(options)
                .toFile(avif)
            );
          }

        }

        await Promise.all(promises);

        console.log("Responsive generada:", file);

      });

      await Promise.all(tasks);

      console.log("✓ Todas las imágenes responsive generadas");

    }

  };

}
